-- Script 3: Sistema SMA Avançado para 1 Minuto
instrument {
    name = 'AM_IQ 1MIN BUY-SELL ADVANCED',
    short_name = '1MIN SMA-ENG',
    icon = 'indicators:BB',
    overlay = true
}

-- Inputs para médias rápidas (otimizadas para 1min)
MaFast_period = input(7, "Ma Fast period", input.integer, 3, 15, 1)
MaFast_average = input(4, "Ma Fast average", input.string_selection, averages.titles)
MaFast_title = input(1, "Ma Fast title", input.string_selection, inputs.titles)

-- Inputs para médias lentas (otimizadas para 1min)
MaSlow_period = input(21, "Ma Slow period", input.integer, 10, 34, 1)
MaSlow_average = input(2, "Ma Slow average", input.string_selection, averages.titles)
MaSlow_title = input(1, "Ma Slow title", input.string_selection, inputs.titles)

-- Inputs para média de tendência (otimizadas para 1min)
MaTrend_period = input(55, "Ma Trend period", input.integer, 34, 89, 1)
MaTrend_average = input(2, "Ma Trend average", input.string_selection, averages.titles)
MaTrend_title = input(1, "Ma Trend title", input.string_selection, inputs.titles)

-- Grupos de input para personalização
input_group {
    "Area Settings",
    colorAreaUp = input { default = "rgba(34, 139, 34, 0.3)", type = input.color },
    colorAreaDown = input { default = "rgba(220, 20, 60, 0.3)", type = input.color },
    visibleArea = input { default = true, type = input.plot_visibility }
}

input_group {
    "Slow Line Settings",
    colorSlow = input { default = "purple", type = input.color },
    widthSlow = input { default = 2, type = input.line_width },
    visibleSlow = input { default = true, type = input.plot_visibility }
}

input_group {
    "Fast Line Settings",
    colorFast = input { default = "blue", type = input.color },
    widthFast = input { default = 1, type = input.line_width },
    visibleFast = input { default = true, type = input.plot_visibility }
}

input_group {
    "Trend Line Settings",
    colorTrend = input { default = "orange", type = input.color },
    widthTrend = input { default = 1, type = input.line_width },
    visibleTrend = input { default = true, type = input.plot_visibility }
}

input_group {
    "Buy Signals",
    colorBuy = input { default = "lime", type = input.color },
    visibleBuy = input { default = true, type = input.plot_visibility }
}

input_group {
    "Sell Signals",
    colorSell = input { default = "red", type = input.color },
    visibleSell = input { default = true, type = input.plot_visibility }
}

input_group {
    "Secondary Buy Signals",
    colorBuy1 = input { default = "green", type = input.color },
    visibleBuy1 = input { default = true, type = input.plot_visibility }
}

input_group {
    "Secondary Sell Signals",
    colorSell1 = input { default = "darkred", type = input.color },
    visibleSell1 = input { default = true, type = input.plot_visibility }
}

input_group {
    "Sniper Buy Signals",
    colorBuy2 = input { default = "lime", type = input.color },
    visibleBuy2 = input { default = true, type = input.plot_visibility }
}

input_group {
    "Sniper Sell Signals",
    colorSell2 = input { default = "red", type = input.color },
    visibleSell2 = input { default = true, type = input.plot_visibility }
}

-- Definindo médias móveis
local avgFast = averages[MaFast_average]
local titleFast = inputs[MaFast_title]

local avgSlow = averages[MaSlow_average]
local titleSlow = inputs[MaSlow_title]

local avgTrend = averages[MaTrend_average]
local titleTrend = inputs[MaTrend_title]

-- Plotando as médias
if visibleFast then
    plot(avgFast(titleFast, MaFast_period), "Ma Fast", colorFast, widthFast)
end

if visibleSlow then
    plot(avgSlow(titleSlow, MaSlow_period), "Ma Slow", colorSlow, widthSlow)
end

if visibleTrend then
    plot(avgTrend(titleTrend, MaTrend_period), "Ma Trend", colorTrend, widthTrend)
end

-- Configuração de timeframes (1min como principal)
candle_time = { "1s", "5s", "10s", "15s", "30s", "1m", "2m", "5m" }
candle_time_res = input(6, "Candle check resolution", input.string_selection, candle_time)

sec = security(current_ticker_id, candle_time[candle_time_res])

filter_source = { "1s", "5s", "10s", "15s", "30s", "1m", "2m", "5m" }
filter_pa_index = input(6, "Filter resolution", input.string_selection, filter_source)

filter_pa = security(current_ticker_id, filter_source[filter_pa_index])

if (sec ~= nil) then
    MaFast0 = avgFast(titleFast, MaFast_period)
    MaFast1 = MaFast0[1]

    MaSlow0 = avgSlow(titleSlow, MaSlow_period)
    MaSlow1 = MaSlow0[1]

    MaTrend0 = avgTrend(titleTrend, MaTrend_period)
    MaTrend1 = MaTrend0[1]

    -- Sinais de compra otimizados para 1min
    if visibleBuy then
        plot_shape((close > open and close[1] < open[1] and close > MaFast0 and MaFast0 > MaSlow0 and MaSlow0 > MaTrend0 and close > open[1] and open <= close[1] and abs(close - open) > abs(close[1] - open[1])),
            "Call",
            shape_style.triangleup,
            shape_size.large,
            colorBuy,
            shape_location.belowbar,
            0,
            "BULL",
            colorBuy
        )
    end

    -- Sinais de venda otimizados para 1min
    if visibleSell then
        plot_shape((close < open and close[1] > open[1] and close < MaFast0 and MaFast0 < MaSlow0 and MaSlow0 < MaTrend0 and close < open[1] and open >= close[1] and abs(close - open) > abs(close[1] - open[1])),
            "Put",
            shape_style.triangledown,
            shape_size.large,
            colorSell,
            shape_location.abovebar,
            0,
            "BEAR",
            colorSell
        )
    end

    -- Sinais de compra secundários
    if visibleBuy1 then
        if (filter_pa.close[1] > filter_pa.open[1] and filter_pa.close[2] < filter_pa.open[2] and filter_pa.close[1] > filter_pa.open[2] and filter_pa.open[1] <= filter_pa.close[2] and abs(filter_pa.close[1] - filter_pa.open[1]) > abs(filter_pa.close[2] - filter_pa.open[2])) then
            plot_shape((close > open and close[1] < open[1] and close > open[1] and open <= close[1] and abs(close - open) > abs(close[1] - open[1])),
                "Call1",
                shape_style.triangleup,
                shape_size.large,
                colorBuy1,
                shape_location.belowbar,
                0,
                "BULL",
                colorBuy1
            )
        end
    end

    -- Sinais de venda secundários
    if visibleSell1 then
        if (filter_pa.close[1] < filter_pa.open[1] and filter_pa.close[2] > filter_pa.open[2] and filter_pa.close[1] < filter_pa.open[2] and filter_pa.open[1] >= filter_pa.close[2] and abs(filter_pa.close[1] - filter_pa.open[1]) > abs(filter_pa.close[2] - filter_pa.open[2])) then
            plot_shape((close < open and close[1] > open[1] and close < open[1] and open >= close[1] and abs(close - open) > abs(close[1] - open[1])),
                "Put1",
                shape_style.triangledown,
                shape_size.large,
                colorSell1,
                shape_location.abovebar,
                0,
                "BEAR",
                colorSell1
            )
        end
    end

    -- Sinais de compra sniper otimizados
    if visibleBuy2 then
        plot_shape((open[2] < close[2] and open[1] > close[1] and close > open and close > close[2]),
            "SNIPER-CALL",
            shape_style.triangleup,
            shape_size.large,
            colorBuy2,
            shape_location.belowbar,
            0,
            "BULL",
            colorBuy2
        )
    end

    -- Sinais de venda sniper otimizados
    if visibleSell2 then
        plot_shape((open[2] > close[2] and open[1] < close[1] and close < open and close < close[2]),
            "SNIPER-PUT",
            shape_style.triangledown,
            shape_size.large,
            colorSell2,
            shape_location.abovebar,
            0,
            "BEAR",
            colorSell2
        )
    end

    -- Área entre as médias
    if visibleArea then
        fill(MaFast0, MaSlow0, "Area", MaFast0 > MaSlow0 and colorAreaUp or MaFast0 < MaSlow0 and colorAreaDown)
    end
end
