--@version=4
study("Estratégia Multi-Indicadores", shorttitle="EMI v1.0", overlay=true)

-- Configuração de Entradas
useSR = input.bool("Usar Suporte/Resistência", true)
useMA = input.bool("Usar Médias Móveis", true)
useMomentum = input.bool("Usar Indicadores de Momento", true)
useCandles = input.bool("Usar Padrões de Candlestick", true)
useBreakouts = input.bool("Usar Breakouts", true)

-- 1. Suporte e Resistência
srPeriod = input(20, "Período para S/R", minval=5)
support = lowest(srPeriod)
resistance = highest(srPeriod)

plot(useSR ? support : na, color=color.green, linewidth=2)
plot(useSR ? resistance : na, color=color.red, linewidth=2)

-- 2. Médias Móveis
maType = input.string("EMA", "Tipo de MA", options={"SMA", "EMA"})
ma1Length = input(9, "MA1 Período")
ma2Length = input(21, "MA2 Período")

calculateMA(src, length, type) =>
    type == "SMA" ? sma(src, length) : ema(src, length)

ma1 = calculateMA(close, ma1Length, maType)
ma2 = calculateMA(close, ma2Length, maType)

plot(useMA ? ma1 : na, color=color.blue, title="MA1")
plot(useMA ? ma2 : na, color=color.orange, title="MA2")

-- 3. Indicadores de Momento (RSI)
rsiLength = input(14, "RSI Período")
overbought = input(70, "Nível Overbought")
oversold = input(30, "Nível Oversold")

rsi = rsi(close, rsiLength)
rsiSignal = crossover(rsi, oversold) or crossunder(rsi, overbought)

-- 4. Padrões de Candlestick
dojiSize = input(0.1, "Tamanho máximo do Doji")
hammerSize = input(0.3, "Tamanho mínimo do Hammer")

isDoji() =>
    math.abs(close - open) <= (high - low) * dojiSize
    
isHammer() =>
    body = math.abs(close - open)
    lowerShadow = low - math.min(open, close)
    upperShadow = high - math.max(open, close)
    body <= (high - low) * hammerSize and lowerShadow >= body * 2

isBullishEngulfing() =>
    close[1] < open[1] and close > open and close > open[1] and open < close[1]

isBearishEngulfing() =>
    close[1] > open[1] and close < open and close < open[1] and open > close[1]

-- 5. Breakouts
breakoutLength = input(5, "Período para Breakout")
upperBreakout = highest(high, breakoutLength)
lowerBreakout = lowest(low, breakoutLength)

-- Condições de Entrada
longCondition = (useMA and crossover(ma1, ma2)) or
               (useMomentum and crossover(rsi, oversold)) or
               (useCandles and (isDoji() or isHammer() or isBullishEngulfing())) or
               (useBreakouts and close > upperBreakout)

shortCondition = (useMA and crossunder(ma1, ma2)) or
                (useMomentum and crossunder(rsi, overbought)) or
                (useCandles and isBearishEngulfing()) or
                (useBreakouts and close < lowerBreakout)

-- Plotagem dos Sinais
plotshape(longCondition and barstate.isclosed, style=shape.triangleup, location=location.belowbar, color=color.green, size=size.small)
plotshape(shortCondition and barstate.isclosed, style=shape.triangledown, location=location.abovebar, color=color.red, size=size.small)

-- Alerta de Confirmação
alertcondition(longCondition, "Sinal de Compra", "Compra")
alertcondition(shortCondition, "Sinal de Venda", "Venda")
