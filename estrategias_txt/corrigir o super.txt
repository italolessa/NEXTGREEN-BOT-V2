instrument {
    name = 'Multi Estratégias - IQ Option',
    short_name = 'MULTI ESTRATEGIAS',
    overlay = true
}

-- Configurações Globais
input_group {
    "Configurações Gerais",
    enableSR = input(true, "Ativar Suporte/Resistência", input.bool),
    enableMA = input(true, "Ativar Médias Móveis", input.bool),
    enableMomentum = input(true, "Ativar Indicadores de Momento", input.bool),
    enableCandlestick = input(true, "Ativar Padrões Candlestick", input.bool),
    enableBreakout = input(true, "Ativar Breakouts", input.bool)
}

-- Parâmetros Suporte/Resistência
input_group {
    "Suporte/Resistência",
    srPeriod = input(50, "Período Lookback", input.integer, 1, 200),
    srColor = input("GRAY", "Cor Linhas", input.color)
}

-- Parâmetros Médias Móveis
input_group {
    "Médias Móveis",
    maType = input("SMA", "Tipo MA", input.string_selection, {"SMA", "EMA"}),
    maPeriod = input(20, "Período MA", input.integer, 1, 100),
    maColor = input("BLUE", "Cor MA", input.color)
}

-- Parâmetros Momentum (RSI)
input_group {
    "Indicador de Momento",
    rsiPeriod = input(14, "Período RSI", input.integer, 1, 50),
    overbought = input(70, "Nível Sobrevendido", input.integer, 1, 100),
    oversold = input(30, "Nível Sobrecomprado", input.integer, 1, 100)
}

-- Funções de Cálculo
local function calculateMA(src, period, maType)
    if maType == "SMA" then
        return sma(src, period)
    else
        return ema(src, period)
    end
end

local function calculateRSI(src, period)
    return rsi(src, period)
end

-- Cálculo Suporte/Resistência
local support = lowest(srPeriod)
local resistance = highest(srPeriod)

-- Cálculo Indicadores
local maValue = calculateMA(close, maPeriod, maType)
local rsiValue = calculateRSI(close, rsiPeriod)

-- Plotagem dos Elementos
if enableSR then
    hline(support, "Suporte", srColor, 1)
    hline(resistance, "Resistência", srColor, 1)
end

if enableMA then
    plot(maValue, "Média Móvel", maColor, 2)
end

if enableMomentum then
    hline(overbought, "Sobrevendido", "GRAY", 1)
    hline(oversold, "Sobrecomprado", "GRAY", 1)
    plot(rsiValue, "RSI", "PURPLE", 2)
end

-- Padrões Candlestick
if enableCandlestick then
    local body = math.abs(close - open)
    local range = high - low
    
    -- Doji
    if body/range < 0.1 then
        plot_shape(shape_style.triangleup, "Doji", "YELLOW", shape_location.belowbar)
    end
    
    -- Martelo
    if (close > open) and ((high - close)/range > 0.6) then
        plot_shape(shape_style.arrowup, "Martelo", "LIME", shape_location.belowbar)
    end
end

-- Detecção Breakouts
if enableBreakout then
    local breakoutUp = crossover(close, resistance)
    local breakoutDown = crossunder(close, support)
    
    if breakoutUp then
        plot_shape(shape_style.arrowup, "Breakout Alta", "LIME", shape_location.belowbar)
    end
    
    if breakoutDown then
        plot_shape(shape_style.arrowdown, "Breakout Baixa", "RED", shape_location.abovebar)
    end
end

-- Configuração de Segurança para Timeframes
candle_time = {"1m", "2m", "5m", "10m", "15m", "30m", "1H"}
check_resolution = input(1, "Resolução Checagem", input.string_selection, candle_time)
sec = security(current_ticker_id, check_resolution)

-- Filtro de Confirmação
if sec ~= nil then
    plot_text("Estratégia Ativa - " .. check_resolution, "status", "GREEN", text_position.bottomright)
end
