    name = 'AM_IQ BUY-SELL SIGNALS',
    short_name = 'BUY-SELL SIGNALS',

}

-- Inputs for Moving Averages
MaFast_period = input(5, "Ma Fast period", input.integer, 1, 1000, 1)
MaValue = input(5, "Ma Value", input.string_selection, inputs.titles)
MaSlow_period = input(20, "Ma Slow period", input.integer, 1, 1000, 1)
Signal_period = input(9, "Signal period", input.integer, 1, 1000, 1)

-- Inputs for Buy/Sell signals
input_group {
    "Buy",
    colorBuy = input { default = "YELLOW", type = input.color }, 
    visibleBuy = input { default = true, type = input.plot_visibility }
}

input_group {
    "Sell",
    colorSell = input { default = "RED", type = input.color },
    visibleSell = input { default = true, type = input.plot_visibility }
}

local titleValue = inputs[MaValue]

-- Calculate Moving Averages
smaFast = sma(titleValue, MaFast_period)
smaSlow = sma(titleValue, MaSlow_period)

-- Calculate differential and weighted moving average
buffer1 = smaFast - smaSlow 
buffer2 = wma(buffer1, Signal_period)

-- Buy/Sell conditions with 99% confidence
confidenceThreshold = 0.99  -- 99% confidence threshold
buyCondition = buffer1 > buffer2 and buffer1[1] < buffer2[1] and close > smaFast and (buffer1 - buffer2) / buffer2 >= confidenceThreshold
sellCondition = buffer1 < buffer2 and buffer1[1] > buffer2[1] and close < smaFast and (buffer2 - buffer1) / buffer1 >= confidenceThreshold

-- Plot Buy/Sell signals
if visibleBuy then
    plot_shape(
    buyCondition,
    "BUY",
    shape_style.arrowup,
    shape_size.large,
    colorBuy,
    shape_location.belowbar,
    -1,
    "BUY",
    colorBuy
)
end

if visibleSell then
    plot_shape(
    sellCondition,
    "SELL",
    shape_style.arrowdown,
    shape_size.large,
    colorSell,
    shape_location.abovebar,
    -1,
    "SELL",
    colorSell
)
end

-- Support/Resistance Levels
instrument { name = "BEAR SUPPORT/RESIST", icon = 'https://bit.ly/3C8cIFQ', overlay = true }

local function calculateSupport()
    local support = make_series()
    local lowLevel = low[2]
    if not get_value(lowLevel) then return support end
    local condition = low >= lowLevel and low[1] >= lowLevel and low[3] >= lowLevel and low[4] >= lowLevel
    support:set(iff(condition, lowLevel, support[1]))
    return support
end

local function calculateResistance()
    local resistance = make_series()
    local highLevel = high[2]
    if not get_value(highLevel) then return resistance end
    local condition = high <= highLevel and high[1] <= highLevel and high[3] <= highLevel and high[4] <= highLevel
    resistance:set(iff(condition, highLevel, resistance[1]))
    return resistance
end

input_group {
    "Color",
    color = input { default = "LIME", type = input.color },
    width = input { default = 1, type = input.line_width }
}

local supportLevel = calculateSupport()
local resistanceLevel = calculateResistance()

hline(supportLevel, "Support", color, width)
hline(resistanceLevel, "Resistance", color, width)
