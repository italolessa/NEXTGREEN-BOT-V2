instrument {
    name = 'Quadrante IQ Option',
    short_name = 'Quadrante',
    overlay = true
}

-- Inputs para cores e visibilidade dos sinais
input_group {
    "Cores e Visibilidade",
    colorQuadrante1 = input { default = "GREEN", type = input.color },  -- Cor do Quadrante 1 (verde)
    colorQuadrante2 = input { default = "BLUE", type = input.color },   -- Cor do Quadrante 2 (azul)
    colorQuadrante3 = input { default = "RED", type = input.color },    -- Cor do Quadrante 3 (vermelho)
    colorQuadrante4 = input { default = "ORANGE", type = input.color }, -- Cor do Quadrante 4 (laranja)
    visibleSinais = input { default = true, type = input.plot_visibility }  -- Visibilidade dos sinais
}

-- Input para o perodo da SMA
input_group {
    "Filtro de Tendncia",
    smaPeriod = input(50, "Perodo da SMA", input.integer, 1, 1000, 1)  -- Perodo da SMA para filtrar tendncia
}

-- Funo para verificar a tendncia (alta ou baixa)
local function checkTrend()
    local smaValue = sma(close, smaPeriod)  -- Calcula a SMA
    if close > smaValue then
        return "ALTA"  -- Tendncia de alta se o preo estiver acima da SMA
    elseif close < smaValue then
        return "BAIXA" -- Tendncia de baixa se o preo estiver abaixo da SMA
    else
        return nil     -- Sem tendncia definida
    end
end

-- Funo para determinar o quadrante
local function determineQuadrant()
    local tendencia = checkTrend()
    local priceChange = close - open

    if tendencia == "ALTA" and priceChange > 0 then
        return 1  -- Quadrante 1: Tendncia de alta e preo subindo
    elseif tendencia == "ALTA" and priceChange < 0 then
        return 2  -- Quadrante 2: Tendncia de alta e preo caindo
    elseif tendencia == "BAIXA" and priceChange < 0 then
        return 3  -- Quadrante 3: Tendncia de baixa e preo caindo
    elseif tendencia == "BAIXA" and priceChange > 0 then
        return 4  -- Quadrante 4: Tendncia de baixa e preo subindo
    else
        return nil  -- Sem quadrante definido
    end
end

-- Determina o quadrante atual
local quadrante = determineQuadrant()

-- Exibe os sinais de quadrante com base na tendncia e no movimento do preo
if visibleSinais then
    if quadrante == 1 then
        plot_shape(
            true,  -- Condio para exibir o sinal
            "Quadrante 1",  -- Nome do sinal
            shape_style.triangleup,  -- Forma do sinal (tringulo para cima)
            shape_size.large,  -- Tamanho do sinal
            colorQuadrante1,  -- Cor do sinal (verde)
            shape_location.belowbar,  -- Localizao do sinal (abaixo da vela)
            0,  -- Offset do sinal
            "Q1",  -- Texto do sinal
            colorQuadrante1  -- Cor do texto
        )
    elseif quadrante == 2 then
        plot_shape(
            true,  -- Condio para exibir o sinal
            "Quadrante 2",  -- Nome do sinal
            shape_style.triangledown,  -- Forma do sinal (tringulo para baixo)
            shape_size.large,  -- Tamanho do sinal
            colorQuadrante2,  -- Cor do sinal (azul)
            shape_location.abovebar,  -- Localizao do sinal (acima da vela)
            0,  -- Offset do sinal
            "Q2",  -- Texto do sinal
            colorQuadrante2  -- Cor do texto
        )
    elseif quadrante == 3 then
        plot_shape(
            true,  -- Condio para exibir o sinal
            "Quadrante 3",  -- Nome do sinal
            shape_style.triangledown,  -- Forma do sinal (tringulo para baixo)
            shape_size.large,  -- Tamanho do sinal
            colorQuadrante3,  -- Cor do sinal (vermelho)
            shape_location.abovebar,  -- Localizao do sinal (acima da vela)
            0,  -- Offset do sinal
            "Q3",  -- Texto do sinal
            colorQuadrante3  -- Cor do texto
        )
    elseif quadrante == 4 then
        plot_shape(
            true,  -- Condio para exibir o sinal
            "Quadrante 4",  -- Nome do sinal
            shape_style.triangleup,  -- Forma do sinal (tringulo para cima)
            shape_size.large,  -- Tamanho do sinal
            colorQuadrante4,  -- Cor do sinal (laranja)
            shape_location.belowbar,  -- Localizao do sinal (abaixo da vela)
            0,  -- Offset do sinal
            "Q4",  -- Texto do sinal
            colorQuadrante4  -- Cor do texto
        )
    end
end
