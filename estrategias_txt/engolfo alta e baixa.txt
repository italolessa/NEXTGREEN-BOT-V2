instrument {
    name = 'ENGOLE ALTA BAIXA',
    short_name = 'ENG',
    icon = 'indicators:BB',
    overlay = true
}

input_group {
    "Buy",
    colorBuy = input { default = "#00FF00", type = input.color },
    visibleBuy = input { default = true, type = input.plot_visibility }
}

input_group {
    "Sell",
    colorSell = input { default = "#FF0000", type = input.color },
    visibleSell = input { default = true, type = input.plot_visibility }
}

plot_shape(
    close > open and close[1] < open[1] and close > high[1],
    "Call",
    shape_style.triangleup,
    shape_size.huge,
    colorBuy,
    shape_location.belowbar,
    0,
    "Eng.Alta",
    colorBuy
)

-- ENGOLE DE BAIXA (VENDA)
plot_shape(
    close < open and close[1] > open[1] and close < low[1],
    "Put",
    shape_style.triangledown,
    shape_size.huge,
    colorSell,
    shape_location.abovebar,
    0,
    "Eng.Baixa",
    colorSell
