instrument {
    name = "CRUZAMENTO+ ENGOLFO",
    short_name = "CMC+ENG",
    overlay = true
}

-- CONFIGURAES PARA CRUZAMENTO DE MDIAS
input_group {
    "SMAA", 
    smaa_color = input{ default = "yellow", type = input.color }
}

input_group {
    "SMAB", 
    smab_color = input{ default = "blue", type = input.color }
}

input_group {
    "Compra Confirmada", 
    up_color = input{ default = "green", type = input.color }
}

input_group {
    "Venda Confirmada", 
    down_color = input{ default = "red", type = input.color }
}

-- CLCULOS PARA CRUZAMENTO DE MDIAS
smaa = ema(close, 5)  -- Mdia rpida (5 perodos)
smab = ema(close, 10) -- Mdia lenta (10 perodos)

-- SINAIS DE COMPRA/VENDA PARA CRUZAMENTO DE MDIAS
cruzamento_compra = (smaa > smab) and (smaa[1] < smab[1])
cruzamento_venda = (smaa < smab) and (smaa[1] > smab[1])

-- SINAIS PARA PADRO ENGOLFO
engolfo_compra = (close[1] < open[1]) and (close > open) and (close > high[1]) and close[2] >= open
engolfo_venda = (close[1] > open[1]) and (close < open) and (close < low[1]) and close[2] <= open

-- SINAIS COMBINADOS (AMBAS AS ESTRATGIAS CONFIRMADAS)
compra_confirmada = cruzamento_compra and engolfo_compra
venda_confirmada = cruzamento_venda and engolfo_venda

-- PLOTAGEM DAS MDIAS
plot(smaa, "Mdia Rpida", smaa_color)
plot(smab, "Mdia Lenta", smab_color)

-- PLOTAGEM DOS SINAIS COMBINADOS (TAMANHO AUMENTADO)
plot_shape(
    compra_confirmada,
    "COMPRA CONFIRMADA",
    shape_style.arrowup,
    shape_size.huge,
    up_color,
    shape_location.belowbar, 
    0,
    "COMPRA CONFIRMADA",
    up_color
)

plot_shape(
    venda_confirmada,
    "VENDA CONFIRMADA",
    shape_style.arrowdown,
    shape_size.huge,
    down_color,
    shape_location.abovebar, 
    0,
    "VENDA CONFIRMADA",
    down_color
)

-- DESTAQUE PARA SINAIS COMBINADOS (CRCULO DE FUNDO)
plot_shape(
    compra_confirmada,
    "DESTAQUE COMPRA",
    shape_style.circle,
    shape_size.large,
    na,
    shape_location.belowbar, 
    0,
    "",
    up_color
)

plot_shape(
    venda_confirmada,
    "DESTAQUE VENDA",
    shape_style.circle,
    shape_size.large,
    na,
    shape_location.abovebar, 
    0,
    "",
    down_color
)

-- ALERTA VISUAL QUANDO AMBOS OS PADRES OCORREM
-- ALERTAS (OPCIONAL, ADICIONADO PARA COMPLETAR)
alertcondition(compra, "Sinal de Compra Detectado", "Compra!")
alertcondition(venda, "Sinal de Venda Detectado", "Venda!")
