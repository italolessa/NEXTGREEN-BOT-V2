}

-- Inputs para cores e visibilidade dos sinais
input_group {
    "Cores e Visibilidade",
    colorQuadrante1 = input { default = "GREEN", type = input.color },  -- Cor do Quadrante 1 (verde)
    colorQuadrante3 = input { default = "RED", type = input.color },    -- Cor do Quadrante 3 (vermelho)
    visibleSinais = input { default = true, type = input.plot_visibility }  -- Visibilidade dos sinais
}

-- Input para o período da SMA
input_group {
    "Filtro de Tendência",
    smaPeriod = input(50, "Período da SMA", input.integer, 1, 1000, 1)  -- Período da SMA para filtrar tendência
}

-- Função para verificar a tendência (alta ou baixa)
local function checkTrend()
    local smaValue = sma(close, smaPeriod)  -- Calcula a SMA
    if close > smaValue then
        return "ALTA"  -- Tendência de alta se o preço estiver acima da SMA
    elseif close < smaValue then
        return "BAIXA" -- Tendência de baixa se o preço estiver abaixo da SMA
    else
        return nil     -- Sem tendência definida
    end
end

-- Função para determinar o quadrante
local function determineQuadrant()
    local tendencia = checkTrend()
    local priceChange = close - open

    if tendencia == "ALTA" and priceChange > 0 then
        return 1  -- Quadrante 1: Tendência de alta e preço subindo
    elseif tendencia == "BAIXA" and priceChange < 0 then
        return 3  -- Quadrante 3: Tendência de baixa e preço caindo
    else
        return nil  -- Ignora outros quadrantes
    end
end

-- Determina o quadrante atual
local quadrante = determineQuadrant()

-- Exibe os sinais de quadrante com base na tendência e no movimento do preço
if visibleSinais then
    if quadrante == 1 then
        plot_shape(
            true,  -- Condição para exibir o sinal
            "Quadrante 1",  -- Nome do sinal
            shape_style.triangleup,  -- Forma do sinal (triângulo para cima)
            shape_size.large,  -- Tamanho do sinal
            colorQuadrante1,  -- Cor do sinal (verde)
            shape_location.belowbar,  -- Localização do sinal (abaixo da vela)
            0,  -- Offset do sinal
            "Q1",  -- Texto do sinal
            colorQuadrante1  -- Cor do texto
        )
    elseif quadrante == 3 then
        plot_shape(
            true,  -- Condição para exibir o sinal
            "Quadrante 3",  -- Nome do sinal
            shape_style.triangledown,  -- Forma do sinal (triângulo para baixo)
            shape_size.large,  -- Tamanho do sinal
            colorQuadrante3,  -- Cor do sinal (vermelho)
            shape_location.abovebar,  -- Localização do sinal (acima da vela)
            0,  -- Offset do sinal
            "Q3",  -- Texto do sinal
            colorQuadrante3  -- Cor do texto
        )
    end
end
