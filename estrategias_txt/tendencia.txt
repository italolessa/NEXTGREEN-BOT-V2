instrument { 
    name = "HackerOB", 
    icon = "indicators:ADX", 
    overlay = true 
}

-- Seleo de Mtodo
method_id = input(1, "Type", input.string_selection, { "TOP" })

-- Configuraes de Nveis
input_group {
    "Maxima",
    level_1_color = input { default = "red", type = input.color },
    level_1_width = input { default = 2, type = input.line_width }
}

input_group {
    "Minima",
    level_2_color = input { default = "green", type = input.color },
    level_2_width = input { default = 2, type = input.line_width }
}

local function m1(candle)
    return candle.high, candle.low
end

local methods = { m1 }

-- Resoluo de 1 minuto
local resolution = "30m"
sec = security(current_ticker_id, resolution)

if sec then
    local method = methods[method_id]
    local c1, c2 = method(sec)
    plot(c1, "C1", level_1_color, level_1_width, 0, style.levels, na_mode.continue)
    plot(c2, "C2", level_2_color, level_2_width, 0, style.levels, na_mode.continue)
end

-- Exibio de Bandas e Mdias Mveis
Exibir_tracamento = input(1, "Deseja exibir o tracamento?", input.string_selection, {"SIM", "NAO"})

input_group {
    "BANDA SUPERIOR", 
    bbsup_color = input { default = "silver", type = input.color }
}

input_group {
    "BANDA INFERIOR", 
    bbinf_color = input { default = "silver", type = input.color }
}

-- Clculo de Bandas e Mdias Mveis
smaa = sma(close, 20)
upper_band = smaa + (stdev(close, 20) * 2.0)  -- Reduzido o multiplicador para 2.0
lower_band = smaa - (stdev(close, 20) * 2.0)
emaa = ema(close, 50)  -- EMA mais curta para 1 minuto

if Exibir_tracamento == 1 then
    plot(emaa, "EMA", emaa_color)
    plot(upper_band, "UPPER_BAND", bbsup_color)
    plot(lower_band, "LOWER_BAND", bbinf_color)
end

-- Configuraes de Mdias Mveis e Perodos
MaFast_period = input(5, "Ma Fast period", input.integer, 1, 1000, 1)  -- Perodo mais curto
MaValue = input(5, "Ma Value", input.string_selection, inputs.titles)
MaSlow_period = input(20, "Ma Slow period", input.integer, 1, 1000, 1)  -- Perodo mais curto
Signal_period = input(5, "Signal period", input.integer, 1, 1000, 1)

-- Configuraes de Compra e Venda
input_group {
    "Compra",
    colorBuy = input { default = "green", type = input.color }, 
    visibleBuy = input { default = true, type = input.plot_visibility }
}

input_group {
    "Venda",
    colorSell = input { default = "red", type = input.color },
    visibleSell = input { default = true, type = input.plot_visibility }
}

-- Clculo de Buffers
smaFast = sma(close, MaFast_period)
smaSlow = sma(close, MaSlow_period)
buffer1 = smaFast - smaSlow 
buffer2 = wma(buffer1, Signal_period)

-- Filtros de Tendncia
trendFilter = sma(close, 30)  -- Mdia mvel de 30 perodos para 1 minuto

-- Filtro de RSI
rsi_value = rsi(close, 10)  -- RSI com perodo mais curto (10)
rsi_filter_buy = rsi_value < 30  -- Filtro de compra (sobrevenda)
rsi_filter_sell = rsi_value > 70  -- Filtro de venda (sobrecompra)

-- Filtro de MACD
macd_line, signal_line, _ = macd(close, 8, 16, 5)  -- MACD com perodos mais curtos
macd_filter_buy = macd_line > signal_line  -- Filtro de compra (MACD acima da linha de sinal)
macd_filter_sell = macd_line < signal_line  -- Filtro de venda (MACD abaixo da linha de sinal)

-- Filtro de ATR para lateralidade
atr_value = atr(10)  -- ATR com perodo mais curto (10)
atr_threshold = input(1.5, "ATR Threshold", input.float)  -- Limite de ATR reduzido
is_sideways = atr_value < atr_threshold  -- Mercado lateralizado se ATR for menor que o limite

-- Filtro de Volume
volume_avg = sma(volume, 10)  -- Mdia mvel do volume em 10 perodos
volume_filter = volume > volume_avg  -- Filtro de volume

-- Filtro de ADX
adx_value = adx(10)  -- ADX com perodo mais curto (10)
adx_filter = adx_value > 25  -- Filtro de ADX (tendncia forte)

-- Filtro de Horrio de Negociao
is_trading_hours = (hour >= 9 and hour <= 17)  -- Exemplo: entre 9h e 17h

-- Condies de Compra e Venda (Simplificadas)
buyCondition = buffer1 > buffer2 and buffer1[1] < buffer2[1] and close > trendFilter
sellCondition = buffer1 < buffer2 and buffer1[1] > buffer2[1] and close < trendFilter

-- Aplicar filtros de lateralidade e ADX
buyCondition = buyCondition and not is_sideways and adx_value
sellCondition = sellCondition and not is_sideways and adx_value

-- Debug das Condies
debug_label("Buy Condition", buyCondition)
debug_label("Sell Condition", sellCondition)

-- Plotagem de Sinais
plot_shape(
    buyCondition and visibleBuy,
    "COMPRA",
    shape_style.triangleup,
    shape_size.large,
    colorBuy,
    shape_location.belowbar,
    -1,
    "COMPRA",
    "green"
) 

plot_shape(
    sellCondition and visibleSell,
    "VENDA",
    shape_style.triangledown,
    shape_size.large,
    colorSell,
    shape_location.abovebar,
    -1,
    "VENDA",
    "red"
)
