instrument {
    name = 'Alerta_5+_Volume_Combinado',
    short_name = 'A5V_Comb',
    overlay = true,
    icon = "indicators:ADX"
}

-- ============================================================
-- PARAMETROS DAS EMAS
-- ============================================================
MaFast_period = input(9, "EMA Rapida", input.integer, 1, 100, 1)
MaSlow_period = input(34, "EMA Lenta", input.integer, 1, 100, 1)

-- ============================================================
-- PARAMETROS DAS 5 CONFIRMAÇÕES
-- ============================================================
ema9_period   = input(9,  "EMA 9 Periodo",   input.integer, 1, 1000, 1)
ema21_period  = input(21, "EMA 21 Periodo",  input.integer, 1, 1000, 1)
ema50_period  = input(50, "EMA 50 Periodo",  input.integer, 1, 1000, 1)
rsi_period    = input(14, "Periodo RSI",     input.integer, 1, 100,  1)

input_group {
    "Cores",
    colorBuy = input { default = "#00FF00", type = input.color },
    colorSell = input { default = "#FF0000", type = input.color },
    colorAlert = input { default = "#FFFF00", type = input.color }
}

-- ============================================================
-- CONFIRMAÇÃO 1: ALINHAMENTO DAS 3 EMAs
-- ============================================================
ema9  = ema(close, ema9_period)
ema21 = ema(close, ema21_period)
ema50 = ema(close, ema50_period)

emas_alta  = ema9 > ema21 and ema21 > ema50
emas_baixa = ema9 < ema21 and ema21 < ema50

-- ============================================================
-- CONFIRMAÇÃO 2: RSI
-- ============================================================
rsi_val  = rsi(close, rsi_period)
rsi_buy  = rsi_val < 70
rsi_sell = rsi_val > 30

-- ============================================================
-- CONFIRMAÇÃO 3: PRESSÃO DA VELA
-- ============================================================
pressao_up   = close > (high + low) / 2
pressao_down = close < (high + low) / 2

-- ============================================================
-- CONFIRMAÇÃO 4: INCLINAÇÃO DAS MÉDIAS
-- ============================================================
inclinacao_buy  = ema9 > ema9[1] and ema21 > ema21[1]
inclinacao_sell = ema9 < ema9[1] and ema21 < ema21[1]

-- ============================================================
-- CONFIRMAÇÃO 5: PREÇO ACIMA/ABAIXO DA EMA 50
-- ============================================================
tendencia_alta  = close > ema50
tendencia_baixa = close < ema50

-- ============================================================
-- CONDIÇÃO DO ALERTA 5+ (TODAS AS 5 CONFIRMAÇÕES)
-- ============================================================
alerta5_call = emas_alta and rsi_buy and pressao_up and inclinacao_buy and tendencia_alta
alerta5_put  = emas_baixa and rsi_sell and pressao_down and inclinacao_sell and tendencia_baixa

-- ============================================================
-- CONDIÇÃO DO VOLUME ATIVO (ENGULFING)
-- ============================================================
-- CALL do Engulfing (Bullish Engulfing)
engulfing_call = (close[1] < open[1]) and 
                 (close > open) and 
                 (close > high[1]) and 
                 (close[1] >= open)

-- PUT do Engulfing (Bearish Engulfing)
engulfing_put = (close[1] > open[1]) and 
                (close < open) and 
                (close < low[1]) and 
                (close[1] <= open)

-- ============================================================
-- SINAL COMBINADO (APENAS QUANDO AMBAS AS CONDIÇÕES SÃO VERDADEIRAS)
-- ============================================================
sinal_call_combinado = alerta5_call and engulfing_call
sinal_put_combinado  = alerta5_put and engulfing_put

-- ============================================================
-- PRÉ-ALERTA (QUANDO FALTA APENAS O ENGULFING)
-- ============================================================
-- Mostra quando tem as 5 confirmações do Alerta5 mas NÃO tem o engulfing ainda
preAlert_call = alerta5_call and not engulfing_call
preAlert_put  = alerta5_put and not engulfing_put

-- ============================================================
-- PLOT DAS EMAS ORIGINAIS
-- ============================================================
emaFast = ema(close, MaFast_period)
emaSlow = ema(close, MaSlow_period)

plot(emaFast, "EMA Rapida", "#FFD700", 1)
plot(emaSlow, "EMA Lenta", "#FFA500", 2)

-- ============================================================
-- PLOT PRÉ-ALERTA (bolinha amarela) - 5 confirmações mas sem engulfing
-- ============================================================
plot_shape(
    preAlert_call,
    "PRE_CALL_AGUARDANDO",
    shape_style.circle,
    shape_size.small,
    colorAlert,
    shape_location.belowbar,
    -1,
    "AGUARDANDO ENGULFING CALL",
    "black"
)

plot_shape(
    preAlert_put,
    "PRE_PUT_AGUARDANDO",
    shape_style.circle,
    shape_size.small,
    colorAlert,
    shape_location.abovebar,
    -1,
    "AGUARDANDO ENGULFING PUT",
    "black"
)

-- ============================================================
-- PLOT SINAL COMBINADO (triângulo) - 5 confirmações + engulfing
-- ============================================================
plot_shape(
    sinal_call_combinado,
    "CALL_COMBINADO",
    shape_style.triangleup,
    shape_size.huge,
    colorBuy,
    shape_location.belowbar,
    0,
    "CALL CONFIRMADO",
    "white"
)

plot_shape(
    sinal_put_combinado,
    "PUT_COMBINADO",
    shape_style.triangledown,
    shape_size.huge,
    colorSell,
    shape_location.abovebar,
    0,
    "PUT CONFIRMADO",
    "white"
)
