instrument {
    name = " Cruz+eng+comb",
    short_name = "CMC+ENG PRO",
    overlay = true
}

-- CONFIGURAES PARA CRUZAMENTO DE MDIAS
input_group {
    "SMAA (Mdia Rpida)", 
    smaa_period = input(5, "Perodo da Mdia Rpida", input.integer, 1),
    smaa_color = input{ default = "yellow", type = input.color }
}

input_group {
    "SMAB (Mdia Lenta)", 
    smab_period = input(10, "Perodo da Mdia Lenta", input.integer, 1),
    smab_color = input{ default = "blue", type = input.color }
}

input_group {
    "Sinais de Cruzamento", 
    cruz_compra_color = input{ default = "lime", type = input.color },
    cruz_venda_color = input{ default = "red", type = input.color }
}

input_group {
    "Sinais Combinados", 
    compra_confirmada_color = input{ default = "green", type = input.color },
    venda_confirmada_color = input{ default = "red", type = input.color }
}

input_group {
    "Configuraes de Engolfo",
    engolfo_compra_color = input{ default = "#00FF00AA", type = input.color },
    engolfo_venda_color = input{ default = "#FF0000AA", type = input.color },
    engolfo_size = input{ default = "Normal", options = {"Pequeno", "Normal", "Grande"}, type = input.options }
}

-- CLCULOS PARA CRUZAMENTO DE MDIAS
smaa = ema(close, smaa_period)  -- Mdia rpida
smab = ema(close, smab_period)  -- Mdia lenta

-- SINAIS DE COMPRA/VENDA PARA CRUZAMENTO DE MDIAS
cruzamento_compra = (smaa > smab) and (smaa[1] < smab[1])
cruzamento_venda = (smaa < smab) and (smaa[1] > smab[1])

-- SINAIS PARA PADRO ENGOLFO
engolfo_compra = (close[1] < open[1]) and (close > open) and (close > high[1]) and close[2] >= open
engolfo_venda = (close[1] > open[1]) and (close < open) and (close < low[1]) and close[2] <= open

-- SINAIS COMBINADOS (AMBAS AS ESTRATGIAS CONFIRMADAS)
compra_confirmada = cruzamento_compra and engolfo_compra
venda_confirmada = cruzamento_venda and engolfo_venda

-- PLOTAGEM DAS MDIAS
plot(smaa, "Mdia Rpida", smaa_color, 2)
plot(smab, "Mdia Lenta", smab_color, 2)

-- DETERMINA TAMANHO DOS SINAIS
local signal_size = shape_size.normal
if engolfo_size == "Pequeno" then
    signal_size = shape_size.small
elseif engolfo_size == "Grande" then
    signal_size = shape_size.large
end

-- PLOTAGEM DOS SINAIS DE CRUZAMENTO (MENOS DESTACADOS)
plot_shape(
    cruzamento_compra and not compra_confirmada,
    "CRUZAMENTO COMPRA",
    shape_style.triangleup,
    shape_size.huge,
    cruz_compra_color,
    shape_location.belowbar, 
    0,
    "CRUZ. COMPRA",
    cruz_compra_color
)

plot_shape(
    cruzamento_venda and not venda_confirmada,
    "CRUZAMENTO VENDA",
    shape_style.triangledown,
    shape_size.huge,
    cruz_venda_color,
    shape_location.abovebar, 
    0,
    "CRUZ. VENDA",
    cruz_venda_color
)

-- PLOTAGEM DOS SINAIS DE ENGOLFO (FUNDO COLORIDO)
plot_shape(
    engolfo_compra and not compra_confirmada,
    "ENGOLFO COMPRA",
    shape_style.circle,
    shape_size.huge,
    engolfo_compra_color,
    shape_location.belowbar, 
    0,
    "ENG. COMPRA",
    engolfo_compra_color
)

plot_shape(
    engolfo_venda and not venda_confirmada,
    "ENGOLFO VENDA",
    shape_style.circle,
    shape_size.huge,
    engolfo_venda_color,
    shape_location.abovebar, 
    0,
    "ENG. VENDA",
    engolfo_venda_color
)

-- PLOTAGEM DOS SINAIS COMBINADOS (MAIS DESTACADOS)
plot_shape(
    compra_confirmada,
    "COMPRA CONFIRMADA",
    shape_style.arrowup,
    shape_size.huge,
    compra_confirmada_color,
    shape_location.belowbar, 
    0,
    "COMPRA CONFIRMADA",
    compra_confirmada_color
)

plot_shape(
    venda_confirmada,
    "VENDA CONFIRMADA",
    shape_style.arrowdown,
    shape_size.huge,
    venda_confirmada_color,
    shape_location.abovebar, 
    0,
    "VENDA CONFIRMADA",
    venda_confirmada_color
)

-- ALERTAS
alertcondition(compra_confirmada, "Sinal de Compra Confirmado", "Compra Confirmada - Cruzamento + Engolfo!")
alertcondition(venda_confirmada, "Sinal de Venda Confirmado", "Venda Confirmada - Cruzamento + Engolfo!")
