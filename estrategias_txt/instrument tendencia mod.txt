instrument {
    name = 'Sinal Baseado na Primeira Vela com Tendncia',
    short_name = 'Primeira Vela + Tendncia',
    overlay = true
}

-- Inputs para cores e visibilidade dos sinais
input_group {
    "Cores e Visibilidade",
    colorCompra = input { default = "BLUE", type = input.color },  -- Cor do sinal de compra (azul)
    colorVenda = input { default = "PINK", type = input.color },   -- Cor do sinal de venda (rosa)
    visibleSinais = input { default = true, type = input.plot_visibility }  -- Visibilidade dos sinais
}

-- Input para o perodo da SMA
input_group {
    "Filtro de Tendncia",
    smaPeriod = input(50, "Perodo da SMA", input.integer, 1, 1000, 1)  -- Perodo da SMA para filtrar tendncia
}

-- Funo para verificar a cor da primeira vela (verde ou vermelha)
local function isCandleGreen(index)
    return close[index] > open[index]  -- Vela verde se o fechamento for maior que a abertura
end

local function isCandleRed(index)
    return close[index] < open[index]  -- Vela vermelha se o fechamento for menor que a abertura
end

-- Funo para verificar a tendncia (alta ou baixa)
local function checkTrend()
    local smaValue = sma(close, smaPeriod)  -- Calcula a SMA
    if close > smaValue then
        return "ALTA"  -- Tendncia de alta se o preo estiver acima da SMA
    elseif close < smaValue then
        return "BAIXA" -- Tendncia de baixa se o preo estiver abaixo da SMA
    else
        return nil     -- Sem tendncia definida
    end
end

-- Verifica a cor da primeira vela
local primeiraVelaVerde = isCandleGreen(1)  -- Primeira vela (ndice 1)
local primeiraVelaVermelha = isCandleRed(1) -- Primeira vela (ndice 1)

-- Verifica a tendncia atual
local tendencia = checkTrend()

-- Exibe os sinais de compra ou venda com base na primeira vela e na tendncia
if visibleSinais then
    if primeiraVelaVerde and tendencia == "ALTA" then
        plot_shape(
            true,  -- Condio para exibir o sinal
            "Compra",  -- Nome do sinal
            shape_style.triangleup,  -- Forma do sinal (tringulo para cima)
            shape_size.large,  -- Tamanho do sinal
            colorCompra,  -- Cor do sinal (azul)
            shape_location.belowbar,  -- Localizao do sinal (abaixo da vela)
            0,  -- Offset do sinal
            "COMPRA",  -- Texto do sinal
            colorCompra  -- Cor do texto
        )
    elseif primeiraVelaVermelha and tendencia == "BAIXA" then
        plot_shape(
            true,  -- Condio para exibir o sinal
            "Venda",  -- Nome do sinal
            shape_style.triangledown,  -- Forma do sinal (tringulo para baixo)
            shape_size.large,  -- Tamanho do sinal
            colorVenda,  -- Cor do sinal (rosa)
            shape_location.abovebar,  -- Localizao do sinal (acima da vela)
            0,  -- Offset do sinal
            "VENDA",  -- Texto do sinal
            colorVenda  -- Cor do texto
        )
    end
end
