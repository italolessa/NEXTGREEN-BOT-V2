instrument {
    name = "GhostForecast Pro",
    short_name = "GHOST",
    overlay = true
}

-- ============================================
-- CONFIGURACOES
-- ============================================
input_group {
    "Sinais de Compra",
    colorBuy = input { default = "#00FF88", type = input.color },
    visibleBuy = input { default = true, type = input.plot_visibility }
}

input_group {
    "Sinais de Venda",
    colorSell = input { default = "#FF4444", type = input.color },
    visibleSell = input { default = true, type = input.plot_visibility }
}

input_group {
    "Configuracoes",
    sensibilidade = input { default = 50, min = 1, max = 100, type = input.integer }
}

-- ============================================
-- MEDIAS PARA TENDENCIA
-- ============================================
ma5 = sma(close, 5)
ma10 = sma(close, 10)
ma20 = sma(close, 20)

-- ============================================
-- RSI (MOMENTUM)
-- ============================================
rsi7 = rsi(close, 7)
rsi14 = rsi(close, 14)

-- ============================================
-- VOLUME
-- ============================================
volume_medio = sma(volume, 20)
volume_alto = volume > volume_medio * 1.2

-- ============================================
-- SINAIS PREDITIVOS (IGUAL SEU SINAL PREDITIVO.TXT)
-- ============================================

-- Gap (abertura)
gap_up = open > high[1]
gap_down = open < low[1]

-- Movimento forte
movimento_buy = (close - open) > (high - low) * 0.6 and close > open
movimento_sell = (open - close) > (high - low) * 0.6 and close < open

-- Sucessao (vela segue a anterior)
sucessao_buy = close > open and close[1] > open[1]
sucessao_sell = close < open and close[1] < open[1]

-- Reversao por RSI extremo
reversao_buy = rsi7 < 30
reversao_sell = rsi7 > 70

-- Engolfo (confirmacao)
engolfo_buy = (close > open and close[1] < open[1] and close > high[1])
engolfo_sell = (close < open and close[1] > open[1] and close < low[1])

-- ============================================
-- PONTUACAO (MESMO DO SEU ARQUIVO)
-- ============================================

pontos_buy = 0
pontos_sell = 0

-- Pontos para COMPRA
if gap_up then pontos_buy = pontos_buy + 25 end
if movimento_buy then pontos_buy = pontos_buy + 35 end
if sucessao_buy then pontos_buy = pontos_buy + 20 end
if reversao_buy then pontos_buy = pontos_buy + 40 end
if volume_alto then pontos_buy = pontos_buy + 15 end
if ma5 > ma10 then pontos_buy = pontos_buy + 10 end

-- Pontos para VENDA
if gap_down then pontos_sell = pontos_sell + 25 end
if movimento_sell then pontos_sell = pontos_sell + 35 end
if sucessao_sell then pontos_sell = pontos_sell + 20 end
if reversao_sell then pontos_sell = pontos_sell + 40 end
if volume_alto then pontos_sell = pontos_sell + 15 end
if ma5 < ma10 then pontos_sell = pontos_sell + 10 end

-- Threshold baseado na sensibilidade
threshold = 60

-- Sinal final
sinal_buy = pontos_buy >= threshold
sinal_sell = pontos_sell >= threshold

-- Hiper confluencia (sinal mais forte)
hiper_buy = sinal_buy and engolfo_buy
hiper_sell = sinal_sell and engolfo_sell

-- ============================================
-- PLOTAGEM DAS MEDIAS
-- ============================================
plot(ma5, "MA5", "#00BFFF", 1)
plot(ma10, "MA10", "#FFA500", 1)
plot(ma20, "MA20", "#888888", 1)

-- ============================================
-- PLOTAGEM DOS SINAIS
-- ============================================

-- SINAL HIPER COMPRA (Estrela)
if visibleBuy and hiper_buy then
    plot_shape(true, "HIPER", shape_style.star, shape_size.huge, 
               "#FFD700", shape_location.belowbar, 0, "HIPER", "#FFD700")
end

-- SINAL HIPER VENDA (Estrela)
if visibleSell and hiper_sell then
    plot_shape(true, "HIPER", shape_style.star, shape_size.huge, 
               "#FFD700", shape_location.abovebar, 0, "HIPER", "#FFD700")
end

-- SINAL COMPRA NORMAL
if visibleBuy and sinal_buy and not hiper_buy then
    plot_shape(true, "COMPRA", shape_style.triangleup, shape_size.huge, 
               colorBuy, shape_location.belowbar, 0, "BUY", colorBuy)
end

-- SINAL VENDA NORMAL
if visibleSell and sinal_sell and not hiper_sell then
    plot_shape(true, "VENDA", shape_style.triangledown, shape_size.huge, 
               colorSell, shape_location.abovebar, 0, "SELL", colorSell)
end

-- ============================================
-- MOSTRAR PONTUACAO NO GRAFICO
-- ============================================
if pontos_buy > 40 then
    plot_shape(true, "F:" .. pontos_buy, shape_style.label, shape_size.small, 
               "#00FF00", shape_location.belowbar, -1, "F", "#00FF00")
end

if pontos_sell > 40 then
    plot_shape(true, "F:" .. pontos_sell, shape_style.label, shape_size.small, 
               "#FF0000", shape_location.abovebar, -1, "F", "#FF0000")
end
