instrument {
    name = 'Quadrante IQ Option (Q1 e Q3)',
    short_name = 'Quadrante Q1/Q3',
    overlay = true
}

-- Inputs Gerais
input_group {
    "Configurações Gerais",
    visibleSinais = input(true, "Exibir Sinais", input.plot_visibility),
    timeframe = input("1", "Timeframe", input.string, list.timeframes)
}

-- Estratégias Disponíveis
input_group {
    "Seleção de Estratégias",
    ativarQuadrante = input(true, "Quadrante Q1/Q3", input.bool),
    ativarSR = input(true, "Suporte/Resistência", input.bool),
    ativarMA = input(true, "Médias Móveis", input.bool),
    ativarMomento = input(true, "Indicador de Momento", input.bool),
    ativarCandle = input(true, "Padrões de Candlestick", input.bool),
    ativarBreakout = input(true, "Breakouts", input.bool)
}

-- Configuração por Estratégia
input_group {
    "1. Quadrante Q1/Q3",
    colorQ1 = input.color(color.green, "Cor Quadrante 1"),
    colorQ3 = input.color(color.red, "Cor Quadrante 3"),
    smaPeriod = input(50, "Período SMA", input.integer, 1)
}

input_group {
    "2. Suporte/Resistência",
    srPeriod = input(20, "Período Lookback", input.integer, 1),
    colorSup = input.color(color.blue, "Cor Suporte"),
    colorRes = input.color(color.orange, "Cor Resistência")
}

input_group {
    "3. Médias Móveis",
    maTipo = input("SMA", "Tipo de Média", input.string, {"SMA", "EMA"}),
    maPeriod = input(20, "Período", input.integer, 1),
    maCor = input.color(color.purple, "Cor da Média")
}

input_group {
    "4. Indicador de Momento",
    rsiPeriod = input(14, "Período RSI", input.integer, 1),
    rsiOB = input(70, "Nível Overbought", input.integer, 1),
    rsiOS = input(30, "Nível Oversold", input.integer, 1),
    rsiCor = input.color(color.yellow, "Cor RSI")
}

input_group {
    "5. Padrões de Candlestick",
    corDoji = input.color(color.white, "Cor Doji"),
    corHammer = input.color(color.blue, "Cor Hammer"),
    corEngulf = input.color(color.purple, "Cor Engulfing")
}

input_group {
    "6. Breakouts",
    brkPeriod = input(20, "Período Lookback", input.integer, 1),
    corBreakout = input.color(color.cyan, "Cor Breakout")
}

-- Funções Comuns
local function calcularSMA()
    return ta.sma(close, smaPeriod)
end

-- 1. Estratégia do Quadrante
local function estrategiaQuadrante()
    local smaValue = calcularSMA()
    local tendencia = close > smaValue and "ALTA" or "BAIXA"
    local movimento = close - open
    
    if tendencia == "ALTA" and movimento > 0 then
        plot_shape(visibleSinais and ativarQuadrante, "Q1", shape.triangleup, shape_size.large, colorQ1, shape_location.belowbar, 0, "Q1", colorQ1)
    elseif tendencia == "BAIXA" and movimento < 0 then
        plot_shape(visibleSinais and ativarQuadrante, "Q3", shape.triangledown, shape_size.large, colorQ3, shape_location.abovebar, 0, "Q3", colorQ3)
    end
end

-- 2. Suporte e Resistência
local function estrategiaSR()
    if ativarSR then
        local sup = ta.lowest(low, srPeriod)
        local res = ta.highest(high, srPeriod)
        plot(sup, "Suporte", colorSup, 1)
        plot(res, "Resistência", colorRes, 1)
    end
end

-- 3. Médias Móveis
local function estrategiaMA()
    if ativarMA then
        local ma = maTipo == "SMA" and ta.sma(close, maPeriod) or ta.ema(close, maPeriod)
        plot(ma, "Média Móvel", maCor, 2)
    end
end

-- 4. Indicador de Momento (RSI)
local function estrategiaMomento()
    if ativarMomento then
        local rsi = ta.rsi(close, rsiPeriod)
        local obSignal = rsi >= rsiOB
        local osSignal = rsi <= rsiOS
        
        plot_shape(obSignal, "RSI OB", shape.triangledown, shape_size.small, rsiCor, shape_location.abovebar, 0, "OB", rsiCor)
        plot_shape(osSignal, "RSI OS", shape.triangleup, shape_size.small, rsiCor, shape_location.belowbar, 0, "OS", rsiCor)
    end
end

-- 5. Padrões de Candlestick
local function estrategiaCandle()
    if ativarCandle then
        -- Doji
        local doji = math.abs(close - open) <= (high - low) * 0.1
        plot_shape(doji, "Doji", shape.circle, shape_size.tiny, corDoji, shape_location.abovebar)
        
        -- Hammer
        local lowerWick = math.min(open, close) - low
        local body = math.abs(close - open)
        local hammer = lowerWick > 2 * body and (high - math.max(open, close)) <= body * 0.3
        plot_shape(hammer, "Hammer", shape.arrowup, shape_size.small, corHammer, shape_location.belowbar)
        
        -- Engulfing
        local engulfBull = close[0] > open[0] and close[1] < open[1] and close[0] > open[1] and open[0] < close[1]
        local engulfBear = close[0] < open[0] and close[1] > open[1] and close[0] < open[1] and open[0] > close[1]
        plot_shape(engulfBull, "Engulf Bull", shape.arrowup, shape_size.small, corEngulf, shape_location.belowbar)
        plot_shape(engulfBear, "Engulf Bear", shape.arrowdown, shape_size.small, corEngulf, shape_location.abovebar)
    end
end

-- 6. Breakouts
local function estrategiaBreakout()
    if ativarBreakout then
        local sup = ta.lowest(low, brkPeriod)
        local res = ta.highest(high, brkPeriod)
        
        local breakoutAlta = close > res
        local breakoutBaixa = close < sup
        
        plot_shape(breakoutAlta, "Breakout Alta", shape.arrowup, shape_size.large, corBreakout, shape_location.belowbar)
        plot_shape(breakoutBaixa, "Breakout Baixa", shape.arrowdown, shape_size.large, corBreakout, shape_location.abovebar)
    end
end

-- Execução Principal
estrategiaQuadrante()
estrategiaSR()
estrategiaMA()
estrategiaMomento()
estrategiaCandle()
estrategiaBreakout()
